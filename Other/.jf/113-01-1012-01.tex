\section*{Discrete}

\subsection*{Uniform}
\begin{itemize}
    \item \textbf{Mean:} \(\frac{a+b}{2}\)
    \item \textbf{Variance:} \(\frac{1}{12}(b-a)(b-a+2)\)
\end{itemize}
\subsection*{Bernouli}
\begin{itemize}
    \item \textbf{Mean:} \(p\)
    \item \textbf{Variance:} \(p(1-p)\)
\end{itemize}
\subsection*{Binomial}
\begin{itemize}
    \item \textbf{Mean:} \(np\) 
    \item \textbf{Variance:} \(np(1-p)\)
\end{itemize}
\subsection*{Geometric}
\begin{itemize}
    \item \textbf{Mean:} \(\frac{1}{p}\)
    \item \textbf{Variance:} \(\frac{1-p}{p^2}\)
\end{itemize}



\section*{Continuous}
\subsection*{Uniform}
\begin{itemize}
    \item \textbf{Mean:} \(\frac{a+b}{2}\)
    \item \textbf{Variance:} \(\frac{(b-a)^2}{12}\)
\end{itemize}
\subsection*{Exponential}
\begin{itemize}
    \item \textbf{Mean:} \(\frac{1}{\lambda}\)
    \item \textbf{Variance:} \(\frac{1}{\lambda^2}\)
    \item \textbf{PDF:} \(\lambda e^{-\lambda x}\)
    \item \textbf{CDF:} \(1- e^{-\lambda x}\)
\end{itemize}
\subsection*{Normal}
\begin{itemize}
    \item \textbf{Mean:} \(\mu\)
    \item \textbf{Variance:} \(\sigma^2\)
\end{itemize}
