\subsection*{Question 1}
\subsubsection*{(a)}
We have
\[
\vec{B}(t) = \frac{\vec{r}\ '(t) \times \vec{r}\ ''(t)}{|\vec{r}\ '(t) \times \vec{r}\ ''(t)|}
\]
Differentiating we get
\[
\frac{d}{dt}(\vec{r}\ ' \times \vec{r}\ '') = \vec{r}\ '' \times \vec{r}\ '' + \vec{r}\ ' \times \vec{r}\ ''' = \vec{r}\ ' \times \vec{r}\ ''',
\]
and $\frac{d}{dt}|\vec{u}| = \frac{\vec{u} \cdot \vec{u}'}{|\vec{u}|}$, then we get
\[
\vec{B}' = \frac{\vec{r}\ ' \times \vec{r}\ '''}{|\vec{r}\ ' \times \vec{r}\ ''|}
- \frac{(\vec{r}\ ' \times \vec{r}\ '')}{|\vec{r}\ ' \times \vec{r}\ ''|^2}
\cdot
\frac{(\vec{r}\ ' \times \vec{r}\ '') \cdot (\vec{r}\ ' \times \vec{r}\ ''')}{|\vec{r}\ ' \times \vec{r}\ ''|}
\]
\[
= \frac{\vec{r}\ ' \times \vec{r}\ '''}{|\vec{r}\ ' \times \vec{r}\ ''|}
- \frac{\bigl[(\vec{r}\ ' \times \vec{r}\ '') \cdot (\vec{r}\ ' \times \vec{r}\ ''')\bigr](\vec{r}\ ' \times \vec{r}\ '')}{|\vec{r}\ ' \times \vec{r}\ ''|^3}.
\]
Calculating \(\vec{N}\) using  $\vec{N} = \vec{B} \times \vec{T}$, 
\[
\vec{N}
= \frac{\vec{r}\ ' \times \vec{r}\ ''}{|\vec{r}\ ' \times \vec{r}\ ''|}
\times \frac{\vec{r}\ '}{|\vec{r}\ '|}
= \frac{(\vec{r}\ ' \times \vec{r}\ '') \times \vec{r}\ '}{|\vec{r}\ '||\vec{r}\ ' \times \vec{r}\ ''|}.
\]
Expanding using property $(\vec{a} \times \vec{b}) \times \vec{c} = (\vec{a} \cdot \vec{c})\vec{b} - (\vec{b} \cdot \vec{c})\vec{a}$:
\[
(\vec{r}\ ' \times \vec{r}\ '') \times \vec{r}\ '
= (\vec{r}\ ' \cdot \vec{r}\ ')\vec{r}\ '' - (\vec{r}\ '' \cdot \vec{r}\ ')\vec{r}\ '
= |\vec{r}\ '|^2 \vec{r}\ '' - (\vec{r}\ ' \cdot \vec{r}\ '')\vec{r}\ ',
\]
\[
\vec{N} = \frac{|\vec{r}\ '|^2 \vec{r}\ '' - (\vec{r}\ ' \cdot \vec{r}\ '')\vec{r}\ '}{|\vec{r}\ '||\vec{r}\ ' \times \vec{r}\ ''|}.
\]
Now we compute $\vec{B}' \cdot \vec{N}$. The second term of $\vec{B}'$ involves
$(\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{N}$, which vanishes since
$\vec{r}\ ' \times \vec{r}\ ''$ is parallel to $\vec{B}$ and $\vec{B}$ is perp to $\vec{N}$.
Therefore
\[
\vec{B}' \cdot \vec{N}
=
\frac{\vec{r}\ ' \times \vec{r}\ '''}{|\vec{r}\ ' \times \vec{r}\ ''|}
\cdot
\frac{|\vec{r}\ '|^2 \vec{r}\ '' - (\vec{r}\ ' \cdot \vec{r}\ '')\vec{r}\ '}
{|\vec{r}\ '||\vec{r}\ ' \times \vec{r}\ ''|}
\]
\[
=
\frac{1}{|\vec{r}\ '||\vec{r}\ ' \times \vec{r}\ ''|^2}
\Big[
(\vec{r}\ ' \times \vec{r}\ ''') \cdot
\bigl(|\vec{r}\ '|^2 \vec{r}\ '' - (\vec{r}\ ' \cdot \vec{r}\ '')\vec{r}\ '\bigr)
\Big].
\]
Since $(\vec{r}\ ' \times \vec{r}\ ''') \cdot \vec{r}\ ' = 0$ (perp)
and $(\vec{r}\ ' \times \vec{r}\ ''') \cdot \vec{r}\ '' 
= (\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ '''$ (property of cross products),
we get
\[
=
-\frac{|\vec{r}\ '|^2 \, (\vec{r}\ ' \times \vec{r}\ ''') \cdot \vec{r}\ ''}
{|\vec{r}\ '||\vec{r}\ ' \times \vec{r}\ ''|^2}
=
-\frac{|\vec{r}\ '| \, (\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ '''}
{|\vec{r}\ ' \times \vec{r}\ ''|^2}.
\]
We have, $\vec{B}' = -\tau|\vec{r}\ '|\vec{N}$,
dotting both sides with the unit vector $\vec{N}$ gives
$\vec{B}' \cdot \vec{N} = -\tau|\vec{r}\ '|$. 
\[
\Rightarrow \tau|\vec{r}\ '|
=
\frac{|\vec{r}\ '| \, (\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ '''}
{|\vec{r}\ ' \times \vec{r}\ ''|^2}
\]
\[
\Rightarrow
\tau
=
\frac{(\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ '''}
{|\vec{r}\ ' \times \vec{r}\ ''|^2}.
\]


\subsubsection*{(b)}

First show $\vec{B}$ is constant, i.e.\ $\vec{B}' = 0$

From we have (a),
\[
\vec{B}' = \frac{\vec{r}\ ' \times \vec{r}\ '''}{|\vec{r}\ ' \times \vec{r}\ ''|}
- \frac{\bigl[(\vec{r}\ ' \times \vec{r}\ '') \cdot (\vec{r}\ ' \times \vec{r}\ ''')\bigr](\vec{r}\ ' \times \vec{r}\ '')}{|\vec{r}\ ' \times \vec{r}\ ''|^3}.
\]

We have $\vec{r}\ '' = \alpha(t)\vec{r}$, so
\[
\vec{r}\ ''' = \alpha'(t)\vec{r} + \alpha(t)\vec{r}\ '.
\]

We compute $\vec{r}\ ' \times \vec{r}\ '''$
\[
\vec{r}\ ' \times \vec{r}\ '''
= \vec{r}\ ' \times \bigl(\alpha'(t)\vec{r} + \alpha(t)\vec{r}\ '\bigr)
= \alpha'(t)(\vec{r}\ ' \times \vec{r}) + \alpha(t)(\vec{r}\ ' \times \vec{r}\ ')
= \alpha'(t)(\vec{r}\ ' \times \vec{r}),
\]
and $\vec{r}\ ' \times \vec{r}\ ''$:
\[
\vec{r}\ ' \times \vec{r}\ ''
= \vec{r}\ ' \times \alpha(t)\vec{r}
= \alpha(t)(\vec{r}\ ' \times \vec{r}).
\]
Therefore $\vec{r}\ ' \times \vec{r}\ '''$ and $\vec{r}\ ' \times \vec{r}\ ''$ are both parallel
to $\vec{r}\ ' \times \vec{r}$,

with $\vec{w} = \vec{r}\ ' \times \vec{r}$ we have
\[
\vec{r}\ ' \times \vec{r}\ '' = \alpha(t)\vec{w}, \qquad \vec{r}\ ' \times \vec{r}\ ''' = \alpha'(t)\vec{w}.
\]
Then
\[
\vec{B}'
= \frac{\alpha'(t)\vec{w}}{|\alpha(t)\vec{w}|}
- \frac{\bigl(\alpha(t)\vec{w} \cdot \alpha'(t)\vec{w}\bigr)\,\alpha(t)\vec{w}}{|\alpha(t)\vec{w}|^3}
= \frac{\alpha'(t)\vec{w}}{|\alpha(t)||\vec{w}|}
- \frac{\alpha(t)\alpha'(t)|\vec{w}|^2 \cdot \alpha(t)\vec{w}}{|\alpha(t)|^3|\vec{w}|^3}.
\]
\[
= \frac{\alpha'(t)\vec{w}}{|\alpha(t)||\vec{w}|}
- \frac{\alpha'(t)\vec{w}}{|\alpha(t)||\vec{w}|}
= \vec{0}.
\]
Hence $\vec{B}$ is constant
\\

Now  compute $\tau(t)$ 
\[
\tau = \frac{(\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ '''}{|\vec{r}\ ' \times \vec{r}\ ''|^2}.
\]
We have $\vec{r}\ ' \times \vec{r}\ '' = \alpha(t)\vec{w}$ and
$\vec{r}\ ''' = \alpha'(t)\vec{r} + \alpha(t)\vec{r}\ '$, so
\[
(\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ '''
= \alpha(t)\vec{w} \cdot \bigl(\alpha'(t)\vec{r} + \alpha(t)\vec{r}\ '\bigr)
= \alpha(t)\alpha'(t)(\vec{r}\ ' \times \vec{r}) \cdot \vec{r}
+ \alpha(t)^2 (\vec{r}\ ' \times \vec{r}) \cdot \vec{r}\ '.
\]
Both terms vanish, since we are dotting a vector with a vector defined to be perpindicular
to it

Hence
\[
(\vec{r}\ ' \times \vec{r}\ '') \cdot \vec{r}\ ''' = 0,
\]
Hence
\[
\tau(t) = 0.
\]
